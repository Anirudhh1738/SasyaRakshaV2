-- ============================================================================
-- SasyaRaksha — Supabase PostgreSQL Full Schema (18 Tables)
-- Safe to run on an empty Supabase PostgreSQL database in Supabase SQL Editor.

-- 1. users
CREATE TABLE IF NOT EXISTS users (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('farmer', 'expert', 'admin')),
  region TEXT DEFAULT 'General Agricultural Zone',
  preferred_language TEXT NOT NULL DEFAULT 'en',
  is_active INTEGER NOT NULL DEFAULT 1,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 2. sessions
CREATE TABLE IF NOT EXISTS sessions (
  token TEXT PRIMARY KEY,
  user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  expires_at TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 3. password_reset_tokens
CREATE TABLE IF NOT EXISTS password_reset_tokens (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  token_hash TEXT NOT NULL UNIQUE,
  expires_at TEXT NOT NULL,
  used_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 4. plants
CREATE TABLE IF NOT EXISTS plants (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  plant_code TEXT NOT NULL UNIQUE,
  crop TEXT NOT NULL,
  field_label TEXT NOT NULL,
  region_label TEXT DEFAULT '',
  owner_id BIGINT REFERENCES users(id) ON DELETE SET NULL,
  is_demo INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 5. weather_records
CREATE TABLE IF NOT EXISTS weather_records (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  user_id BIGINT REFERENCES users(id) ON DELETE SET NULL,
  location_name TEXT NOT NULL,
  latitude DOUBLE PRECISION NOT NULL,
  longitude DOUBLE PRECISION NOT NULL,
  crop TEXT NOT NULL,
  temperature_c DOUBLE PRECISION NOT NULL,
  humidity_pct DOUBLE PRECISION NOT NULL,
  rainfall_mm DOUBLE PRECISION NOT NULL,
  risk_level TEXT NOT NULL CHECK (risk_level IN ('Low', 'Medium', 'High')),
  risk_factors_json TEXT NOT NULL DEFAULT '[]',
  time_period TEXT NOT NULL DEFAULT '',
  is_demo INTEGER NOT NULL DEFAULT 0,
  recorded_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 6. predictions
CREATE TABLE IF NOT EXISTS predictions (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  plant_id BIGINT REFERENCES plants(id) ON DELETE SET NULL,
  user_id BIGINT REFERENCES users(id) ON DELETE SET NULL,
  weather_record_id BIGINT REFERENCES weather_records(id) ON DELETE SET NULL,
  crop TEXT NOT NULL,
  predicted_disease TEXT NOT NULL,
  condition_category TEXT NOT NULL DEFAULT 'fungal',
  confidence DOUBLE PRECISION NOT NULL,
  uncertain INTEGER NOT NULL DEFAULT 0,
  uncertainty_reason TEXT DEFAULT '',
  alternative_predictions_json TEXT NOT NULL DEFAULT '[]',
  affected_area_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  severity TEXT NOT NULL CHECK (severity IN ('Healthy', 'Low', 'Medium', 'High')),
  leaf_coverage_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  chlorosis_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  necrosis_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  original_image_path TEXT NOT NULL,
  processed_image_path TEXT NOT NULL,
  model_engine TEXT NOT NULL,
  verification_status TEXT NOT NULL DEFAULT 'pending' CHECK (verification_status IN ('pending', 'confirmed', 'rejected', 'corrected')),
  verified_crop TEXT,
  verified_disease TEXT,
  region_label TEXT DEFAULT '',
  approx_lat DOUBLE PRECISION,
  approx_lng DOUBLE PRECISION,
  notes TEXT DEFAULT '',
  symptoms_json TEXT NOT NULL DEFAULT '[]',
  symptom_onset_value INTEGER,
  symptom_onset_unit TEXT DEFAULT '',
  image_consistency TEXT NOT NULL DEFAULT 'single',
  consistency_message TEXT DEFAULT '',
  user_feedback_status TEXT NOT NULL DEFAULT 'unreviewed',
  user_feedback_notes TEXT DEFAULT '',
  same_plant_confirmed INTEGER,
  recommendations_json TEXT NOT NULL DEFAULT '[]',
  is_demo INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 7. screening_images
CREATE TABLE IF NOT EXISTS screening_images (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  prediction_id BIGINT NOT NULL REFERENCES predictions(id) ON DELETE CASCADE,
  image_role TEXT NOT NULL DEFAULT 'closeup_leaf',
  original_path TEXT NOT NULL,
  processed_path TEXT NOT NULL DEFAULT '',
  suitable INTEGER NOT NULL DEFAULT 1,
  rejection_reason TEXT DEFAULT '',
  detected_crop TEXT NOT NULL DEFAULT '',
  detected_disease TEXT NOT NULL DEFAULT '',
  confidence DOUBLE PRECISION NOT NULL DEFAULT 0,
  affected_area_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  severity TEXT NOT NULL DEFAULT 'Healthy',
  mime_type TEXT NOT NULL DEFAULT 'image/jpeg',
  file_size_bytes INTEGER NOT NULL DEFAULT 0,
  width INTEGER NOT NULL DEFAULT 0,
  height INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 8. soil_records
CREATE TABLE IF NOT EXISTS soil_records (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

  observed_moisture TEXT DEFAULT '',
  observed_texture TEXT DEFAULT '',
  observed_surface_signs TEXT DEFAULT '',
  soil_disclaimer TEXT DEFAULT '',
  manual_moisture TEXT DEFAULT '',
  manual_ph TEXT DEFAULT '',
  manual_soil_type TEXT DEFAULT '',

  recommendations_json TEXT NOT NULL DEFAULT '[]',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()

-- 9. uploaded_images
CREATE TABLE IF NOT EXISTS uploaded_images (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  prediction_id BIGINT NOT NULL UNIQUE REFERENCES predictions(id) ON DELETE CASCADE,
  original_path TEXT NOT NULL,
  processed_path TEXT NOT NULL,

  file_size_bytes INTEGER NOT NULL,

-- 10. severity_analyses
CREATE TABLE IF NOT EXISTS severity_analyses (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  prediction_id BIGINT NOT NULL UNIQUE REFERENCES predictions(id) ON DELETE CASCADE,
  affected_area_pct DOUBLE PRECISION NOT NULL,
  severity TEXT NOT NULL CHECK (severity IN ('Healthy', 'Low', 'Medium', 'High')),
  leaf_coverage_pct DOUBLE PRECISION NOT NULL,
  chlorosis_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  necrosis_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  rust_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  powdery_pct DOUBLE PRECISION NOT NULL DEFAULT 0,
  leaf_pixels INTEGER NOT NULL DEFAULT 0,
  infected_pixels INTEGER NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 11. field_passports
CREATE TABLE IF NOT EXISTS field_passports (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  passport_code TEXT NOT NULL UNIQUE,
  prediction_id BIGINT NOT NULL UNIQUE REFERENCES predictions(id) ON DELETE CASCADE,

-- 14. treatment_records

-- 15. progression_entries

-- 16. expert_corrections

  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  prediction_id BIGINT NOT NULL REFERENCES predictions(id) ON DELETE CASCADE,
  reviewer_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  reviewer_name TEXT NOT NULL,
  action TEXT NOT NULL CHECK (action IN ('confirm', 'reject', 'correct')),

  actor_role TEXT NOT NULL DEFAULT 'system',

  details_json TEXT NOT NULL DEFAULT '{}',
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()

-- 18. model_evaluations
CREATE TABLE IF NOT EXISTS model_evaluations (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  model_name TEXT NOT NULL,
  dataset_name TEXT NOT NULL,
  status TEXT NOT NULL CHECK (status IN ('not_evaluated', 'evaluated')),
  accuracy DOUBLE PRECISION,
  precision_macro DOUBLE PRECISION,
  recall_macro DOUBLE PRECISION,
  f1_macro DOUBLE PRECISION,
  confusion_matrix_json TEXT DEFAULT '[]',
  classes_json TEXT DEFAULT '[]',
  notes TEXT DEFAULT '',
  evaluated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Indexes for user-isolated queries, progression tracking & outbreak filters
CREATE INDEX IF NOT EXISTS idx_sessions_user ON sessions(user_id);

CREATE INDEX IF NOT EXISTS idx_predictions_plant ON predictions(plant_id);
CREATE INDEX IF NOT EXISTS idx_screening_images_pred ON screening_images(prediction_id);
CREATE INDEX IF NOT EXISTS idx_plants_owner ON plants(owner_id);
CREATE INDEX IF NOT EXISTS idx_treatment_plant ON treatment_records(plant_id, user_id);
CREATE INDEX IF NOT EXISTS idx_progression_plant ON progression_entries(plant_id, user_id);

CREATE INDEX IF NOT EXISTS idx_expert_corrections_pred ON expert_corrections(prediction_id);
CREATE INDEX IF NOT EXISTS idx_audit_logs_created ON audit_logs(created_at DESC);
