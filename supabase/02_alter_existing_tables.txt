-- 6. Ensure UNIQUE constraint on disease_library(crop, disease_name) if missing
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conrelid = 'disease_library'::regclass
      AND contype = 'u'
  ) THEN
    ALTER TABLE disease_library ADD CONSTRAINT disease_library_crop_disease_unique UNIQUE (crop, disease_name);
  END IF;
EXCEPTION
  WHEN undefined_table THEN NULL;
END $$;

-- 7. Synchronize identity sequences safely (useful after manual/CSV/script imports)
DO $$
DECLARE
  tbl TEXT;
BEGIN
  FOREACH tbl IN ARRAY ARRAY[
    'users', 'password_reset_tokens', 'plants', 'weather_records', 'predictions',
    'screening_images', 'soil_records', 'uploaded_images', 'severity_analyses',
    'field_passports', 'geographical_reports', 'disease_library', 'treatment_records',
    'progression_entries', 'expert_corrections', 'audit_logs', 'model_evaluations'
  ]
  LOOP
    EXECUTE format(
      'SELECT setval(pg_get_serial_sequence(%L, ''id''), COALESCE((SELECT MAX(id) FROM %I), 1), true)',
      tbl, tbl
    );
  END LOOP;
END $$;
